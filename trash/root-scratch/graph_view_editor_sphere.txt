    <script>
        window.onload = function() {
            lucide.createIcons();

            // 1. 에디터 설정 (Quill)
            var toolbarOptions = [
                [{ 'header': [1, 2, 3, false] }],
                ['bold', 'italic', 'underline', 'strike'],
                [{ 'script': 'sub'}, { 'script': 'super' }],
                ['blockquote', 'code-block'],
                [{ 'list': 'ordered'}, { 'list': 'bullet' }],
                [{ 'align': [] }],
                ['link', 'image', 'formula'], 
                ['clean']
            ];

            var quill = new Quill('#editor-container', {
                modules: { formula: true, toolbar: toolbarOptions },
                theme: 'snow',
                placeholder: '이 주제에 대한 내용을 작성하세요. 수식(fx) 및 이미지 첨부가 가능합니다.'
            });

            const documentStore = {};
            let currentTopicName = null;

            function openEditor(topicName) {
                // 이전 내용 저장
                if (currentTopicName !== null) documentStore[currentTopicName] = quill.root.innerHTML;
                
                currentTopicName = topicName;
                document.getElementById('editor-title').innerText = topicName.replace('\n', ' ');

                if (documentStore[topicName]) {
                    quill.root.innerHTML = documentStore[topicName];
                } else {
                    quill.root.innerHTML = `<h1>${topicName.replace('\n', ' ')}</h1><p><br></p>`;
                }

                document.getElementById('editor-panel').classList.add('open');
            }

            function closeEditor() {
                if (currentTopicName !== null) documentStore[currentTopicName] = quill.root.innerHTML;
                document.getElementById('editor-panel').classList.remove('open');
                currentTopicName = null;
            }

            document.getElementById('close-editor-btn').addEventListener('click', closeEditor);
            quill.on('text-change', function() {
                if (currentTopicName !== null) documentStore[currentTopicName] = quill.root.innerHTML;
            });


            // 2. Three.js 설정
            const container = document.getElementById('canvas-container');
            const graphContainer = document.getElementById('graph-container');
            const scene = new THREE.Scene();
            
            let canvasWidth = graphContainer.clientWidth;
            let canvasHeight = graphContainer.clientHeight;

            const camera = new THREE.PerspectiveCamera(60, canvasWidth / canvasHeight, 0.1, 1000);
            camera.position.set(0, 0, 60);

            const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
            renderer.setSize(canvasWidth, canvasHeight);
            renderer.setPixelRatio(window.devicePixelRatio);
            container.appendChild(renderer.domElement);

            const masterRotationGroup = new THREE.Group();
            scene.add(masterRotationGroup);
            
            const masterTranslationGroup = new THREE.Group();
            masterRotationGroup.add(masterTranslationGroup);

            masterTranslationGroup.userData.sphereNodes = [];

            let targetZoomZ = 60; 
            let targetNavZ = 0;   
            const masterTargetQuat = new THREE.Quaternion(); 

            // 배경 파티클
            const particleGeometry = new THREE.BufferGeometry();
            const particleCount = 2000;
            const posArray = new Float32Array(particleCount * 3);
            for(let i = 0; i < particleCount * 3; i++) posArray[i] = (Math.random() - 0.5) * 400;
            particleGeometry.setAttribute('position', new THREE.BufferAttribute(posArray, 3));
            const particleMaterial = new THREE.PointsMaterial({ size: 0.15, color: 0xffffff, transparent: true, opacity: 0.15, blending: THREE.AdditiveBlending });
            const particlesMesh = new THREE.Points(particleGeometry, particleMaterial);
            masterTranslationGroup.add(particlesMesh);

            const wireframeMaterial = new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.2, blending: THREE.AdditiveBlending });
            const nodeGeometry = new THREE.SphereGeometry(0.2, 16, 16);
            const baseNodeMaterial = new THREE.MeshBasicMaterial({ color: 0xaaaaaa, transparent: true, opacity: 0.6 });
            const lineMaterial = new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.45 }); 

            const raycaster = new THREE.Raycaster();
            const mouse = new THREE.Vector2(-1, -1);
            
            const nodesArray = [];     // 호버 효과용 작은 노드들
            const hitboxesArray = [];  // 클릭/드래그용 큰 매니폴드 영역
            const allSphereGroups = [];
            const groupUpdaters = [];

            let isFlattened = false;
            let globalFlattenProgress = 0;
            const identityQuat = new THREE.Quaternion(0, 0, 0, 1);
            const clamp = (val, min, max) => Math.min(Math.max(val, min), max);

            // Sphere Grid 생성 로직
            function createSphereGrid(sphereRadius, basePos, flatPos, nodeCount, labelText, groupIndex) {
                const group = new THREE.Group();
                group.userData.baseGroupPos = new THREE.Vector3(...basePos);
                group.userData.flatGroupPos = new THREE.Vector3(...flatPos);
                group.position.copy(group.userData.baseGroupPos);
                group.rotation.set(Math.random() * Math.PI, Math.random() * Math.PI, Math.random() * Math.PI);
                group.userData.baseQuaternion = group.quaternion.clone();

                const segments = Math.max(20, Math.floor(40 * (sphereRadius / 12))); 
                const geometry = new THREE.SphereGeometry(sphereRadius, segments, segments);
                
                const indices = [];
                for (let y = 0; y <= segments; y++) {
                    for (let x = 0; x <= segments; x++) {
                        const i = y * (segments + 1) + x;
                        if (x < segments) indices.push(i, i + 1);
                        if (y < segments) indices.push(i, i + segments + 1);
                    }
                }
                geometry.setIndex(indices);

                const positions = geometry.attributes.position;
                const uvs = geometry.attributes.uv; 
                const basePosArray = new Float32Array(positions.count * 3);
                const flatPosArray = new Float32Array(positions.count * 3);
                const flatGridRadius = 12;

                for (let i = 0; i < positions.count; i++) {
                    const x = positions.getX(i); const y = positions.getY(i); const z = positions.getZ(i);
                    basePosArray[i*3] = x; basePosArray[i*3+1] = y; basePosArray[i*3+2] = z;
                    const u = uvs.getX(i); const v = uvs.getY(i); 
                    const flatX = (u - 0.5) * (flatGridRadius * 4.5); 
                    const flatY = (v - 0.5) * (flatGridRadius * 2.2); 
                    flatPosArray[i*3] = flatX; flatPosArray[i*3+1] = flatY; flatPosArray[i*3+2] = 0;
                }
                
                geometry.setAttribute('basePosition', new THREE.BufferAttribute(basePosArray, 3));
                geometry.setAttribute('flatPosition', new THREE.BufferAttribute(flatPosArray, 3));
                
                const manifold = new THREE.LineSegments(geometry, wireframeMaterial);
                group.add(manifold);

                // 큰 히트박스 (주제 클릭 및 드래그용)
                const hitGeometry = new THREE.SphereGeometry(sphereRadius, 16, 16);
                const hitbox = new THREE.Mesh(hitGeometry, new THREE.MeshBasicMaterial({ transparent: true, opacity: 0 }));
                hitbox.userData.topicName = labelText || "UNNAMED TOPIC"; // 에디터 오픈 시 활용
                group.add(hitbox);
                hitboxesArray.push(hitbox);

                const groupNodesArray = [];
                const nodePositions = [];
                const selectedIndices = new Set();
                while (selectedIndices.size < nodeCount) selectedIndices.add(Math.floor(Math.random() * positions.count));

                const vertex = new THREE.Vector3();
                selectedIndices.forEach(index => {
                    vertex.fromBufferAttribute(positions, index);
                    const nodeMesh = new THREE.Mesh(nodeGeometry, baseNodeMaterial.clone());
                    nodeMesh.position.copy(vertex);
                    
                    // 호버 효과 데이터 할당
                    nodeMesh.userData = { 
                        baseColor: 0xaaaaaa, hoverColor: 0xffffff, baseScale: 1.0, hoverScale: 2.0,
                        basePos: vertex.clone(),
                        flatPos: new THREE.Vector3().fromArray(flatPosArray, index * 3)
                    };
                    
                    group.add(nodeMesh);
                    nodesArray.push(nodeMesh); 
                    groupNodesArray.push(nodeMesh);
                    nodePositions.push(nodeMesh.userData.basePos);
                });

                masterTranslationGroup.userData.sphereNodes.push(groupNodesArray);

                const edges = [];
                nodePositions.forEach((pos1, i) => {
                    const distances = [];
                    nodePositions.forEach((pos2, j) => {
                        if (i !== j) distances.push({ index: j, distance: pos1.distanceTo(pos2) });
                    });
                    distances.sort((a, b) => a.distance - b.distance);
                    for (let k = 0; k < 2; k++) {
                        if (distances[k]) edges.push({ nodeA: groupNodesArray[i], nodeB: groupNodesArray[distances[k].index] });
                    }
                });

                const lineGeometry = new THREE.BufferGeometry();
                const linePosArray = new Float32Array(edges.length * 6);
                lineGeometry.setAttribute('position', new THREE.BufferAttribute(linePosArray, 3));
                const networkLines = new THREE.LineSegments(lineGeometry, lineMaterial);
                group.add(networkLines);

                if (labelText) {
                    const canvas = document.createElement('canvas');
                    const context = canvas.getContext('2d');
                    canvas.width = 512; canvas.height = 256;
                    context.font = "300 32px 'Helvetica Neue', Arial, sans-serif";
                    context.fillStyle = "rgba(255, 255, 255, 1.0)"; 
                    context.textAlign = "center"; context.textBaseline = "middle";
                    if ('letterSpacing' in context) context.letterSpacing = "6px";
                    const lines = labelText.split('\n');
                    const lineHeight = 44;
                    const startY = (canvas.height / 2) - ((lines.length - 1) * lineHeight / 2);
                    for (let i = 0; i < lines.length; i++) context.fillText(lines[i].toUpperCase(), canvas.width / 2, startY + i * lineHeight);
                    
                    const sprite = new THREE.Sprite(new THREE.SpriteMaterial({ map: new THREE.CanvasTexture(canvas), transparent: true, depthTest: false, depthWrite: false }));
                    sprite.renderOrder = 999;
                    const scale = Math.max(10, flatGridRadius * 1.5);
                    sprite.scale.set(scale, scale / 2, 1); 
                    sprite.position.z = 1.5; 
                    group.add(sprite); 
                }

                masterTranslationGroup.add(group);
                allSphereGroups.push(group);

                const delayOffset = groupIndex * 0.12; 
                groupUpdaters.push((globalProg) => {
                    const localProg = clamp((globalProg - delayOffset) * 2.5, 0, 1);
                    group.position.lerpVectors(group.userData.baseGroupPos, group.userData.flatGroupPos, localProg);

                    const posAttr = manifold.geometry.attributes.position;
                    const baseAttr = manifold.geometry.attributes.basePosition;
                    const flatAttr = manifold.geometry.attributes.flatPosition;
                    for(let i=0; i < posAttr.count; i++) {
                        posAttr.setXYZ(i,
                            THREE.MathUtils.lerp(baseAttr.getX(i), flatAttr.getX(i), localProg),
                            THREE.MathUtils.lerp(baseAttr.getY(i), flatAttr.getY(i), localProg),
                            THREE.MathUtils.lerp(baseAttr.getZ(i), flatAttr.getZ(i), localProg)
                        );
                    }
                    posAttr.needsUpdate = true;

                    for(let i=0; i < groupNodesArray.length; i++) {
                        const node = groupNodesArray[i];
                        node.position.lerpVectors(node.userData.basePos, node.userData.flatPos, localProg);
                    }

                    const edgePosAttr = networkLines.geometry.attributes.position;
                    let idx = 0;
                    for(let i=0; i < edges.length; i++) {
                        const a = edges[i].nodeA.position; const b = edges[i].nodeB.position;
                        edgePosAttr.setXYZ(idx++, a.x, a.y, a.z); edgePosAttr.setXYZ(idx++, b.x, b.y, b.z);
                    }
                    edgePosAttr.needsUpdate = true;

                    group.quaternion.copy(group.userData.baseQuaternion).slerp(identityQuat, localProg);
                    hitbox.scale.set(1, 1, 1 - (localProg * 0.95));
                });
            }

            const zSpacing = -45;
            createSphereGrid(12,  [  0,   0,   0], [0, 0, zSpacing * 0], 20, "PHYSICS", 0);
            createSphereGrid( 7,  [-22,  12, -10], [0, 0, zSpacing * 1], 12, "QUANTUM\nMECHANICS", 1);
            createSphereGrid( 8,  [ 24, -10,   5], [0, 0, zSpacing * 2], 15, "CLASSICAL\nMECHANICS", 2);
            createSphereGrid( 5.5,[-15, -18, -25], [0, 0, zSpacing * 3], 10, "STATISTICAL\nMECHANICS", 3);
            createSphereGrid( 4.5,[ 18,  20, -15], [0, 0, zSpacing * 4],  8, "ELECTRO\nDYNAMICS", 4);
            createSphereGrid( 6,  [  0,  25, -20], [0, 0, zSpacing * 5], 10, "MATHEMATICAL\nPHYSICS", 5);

            const interSphereConnections = [];
            const fibersPerLayer = 10; 
            const sphereNodes = masterTranslationGroup.userData.sphereNodes;
            for (let i = 0; i < sphereNodes.length - 1; i++) {
                const nodesA = sphereNodes[i]; const nodesB = sphereNodes[i+1];
                for (let j = 0; j < fibersPerLayer; j++) {
                    const nodeA = nodesA[Math.floor(Math.random() * nodesA.length)];
                    const nodeB = nodesB[Math.floor(Math.random() * nodesB.length)];
                    const randomDir = new THREE.Vector3(Math.random() - 0.5, Math.random() - 0.5, Math.random() - 0.5).normalize();
                    interSphereConnections.push({ nodeA, nodeB, randomDir });
                }
            }
            
            const segmentsPerFiber = 20; 
            const totalFiberPoints = interSphereConnections.length * segmentsPerFiber * 2; 
            const fiberPositions = new Float32Array(totalFiberPoints * 3);
            const fiberGeometry = new THREE.BufferGeometry();
            fiberGeometry.setAttribute('position', new THREE.BufferAttribute(fiberPositions, 3));
            const fiberMaterial = new THREE.LineBasicMaterial({ color: 0xeeeeee, transparent: true, opacity: 0.4, blending: THREE.AdditiveBlending });
            const fiberLines = new THREE.LineSegments(fiberGeometry, fiberMaterial);
            fiberLines.frustumCulled = false; 
            masterTranslationGroup.add(fiberLines); 

            const tempFiberCurve = new THREE.QuadraticBezierCurve3();
            const tempPosA = new THREE.Vector3();
            const tempPosB = new THREE.Vector3();

            // 상호작용 이벤트 바인딩
            graphContainer.addEventListener('wheel', (e) => {
                targetZoomZ += e.deltaY * 0.05;
                targetZoomZ = clamp(targetZoomZ, 20, 150); 
            }, { passive: true });

            const zScrollbar = document.getElementById('z-scrollbar');
            zScrollbar.addEventListener('input', (e) => { if(isFlattened) targetNavZ = parseFloat(e.target.value); });

            const mapBtn = document.getElementById('exp-map-btn');
            const scrollbarContainer = document.getElementById('z-scrollbar-container');

            // --- 스크롤바 플로팅 패널 드래그 로직 추가 ---
            let isDraggingScrollbar = false;
            let scrollbarOffsetX = 0;
            let scrollbarOffsetY = 0;

            scrollbarContainer.addEventListener('pointerdown', (e) => {
                if (e.target.id === 'z-scrollbar') return; // 슬라이더 자체를 조작할 때는 패널 이동 방지
                isDraggingScrollbar = true;
                const rect = scrollbarContainer.getBoundingClientRect();
                scrollbarOffsetX = e.clientX - rect.left;
                scrollbarOffsetY = e.clientY - rect.top;
                scrollbarContainer.style.cursor = 'grabbing';
                e.stopPropagation();
            });

            window.addEventListener('pointermove', (e) => {
                if (isDraggingScrollbar) {
                    scrollbarContainer.style.right = 'auto'; 
                    scrollbarContainer.style.bottom = 'auto';
                    scrollbarContainer.style.left = `${e.clientX - scrollbarOffsetX}px`;
                    scrollbarContainer.style.top = `${e.clientY - scrollbarOffsetY}px`;
                    scrollbarContainer.style.transform = 'none'; // 중앙 정렬 설정(translateY) 해제
                }
            });

            window.addEventListener('pointerup', () => {
                if (isDraggingScrollbar) {
                    isDraggingScrollbar = false;
                    scrollbarContainer.style.cursor = 'grab';
                }
            });
            // ---------------------------------------------

            mapBtn.addEventListener('click', () => {
                isFlattened = !isFlattened;
                if(isFlattened) {
                    mapBtn.innerHTML = 'EXPONENTIAL MAP [ ON ]'; mapBtn.classList.add('active');
                    document.getElementById('mode-subtitle').innerText = "Z-AXIS MANIFOLD EXPLORATION";
                    targetZoomZ = 60; targetNavZ = 0; zScrollbar.value = 0; masterTargetQuat.identity();
                    scrollbarContainer.classList.add('visible');
                } else {
                    mapBtn.innerHTML = 'EXPONENTIAL MAP [ OFF ]'; mapBtn.classList.remove('active');
                    document.getElementById('mode-subtitle').innerText = "SCATTERED TOPOLOGICAL MANIFOLDS";
                    targetZoomZ = 60; targetNavZ = 0; zScrollbar.value = 0; masterTargetQuat.identity();
                    scrollbarContainer.classList.remove('visible');
                }
            });

            scene.fog = new THREE.Fog(0x000000, 20, 120);

            let hoveredNode = null;
            let isDraggingSphere = false;
            let isDraggingMaster = false;
            let draggedGroup = null;
            let pointerDownPos = { x: 0, y: 0 };
            let previousMousePosition = { x: 0, y: 0 };

            renderer.domElement.addEventListener('pointerdown', (event) => {
                const rect = renderer.domElement.getBoundingClientRect();
                mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
                mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
                
                raycaster.setFromCamera(mouse, camera);
                
                if (!isFlattened) {
                    const intersects = raycaster.intersectObjects(hitboxesArray);
                    if (intersects.length > 0) {
                        isDraggingSphere = true;
                        draggedGroup = intersects[0].object.parent;
                        container.style.cursor = 'grabbing';
                    } else {
                        isDraggingMaster = true;
                        container.style.cursor = 'grabbing';
                    }
                } else {
                    isDraggingMaster = true;
                    container.style.cursor = 'grabbing';
                }
                pointerDownPos = { x: event.clientX, y: event.clientY };
                previousMousePosition = { x: event.clientX, y: event.clientY };
            });

            renderer.domElement.addEventListener('pointermove', (event) => {
                const rect = renderer.domElement.getBoundingClientRect();
                mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
                mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;

                const deltaX = event.clientX - previousMousePosition.x;
                const deltaY = event.clientY - previousMousePosition.y;

                const dist = Math.hypot(deltaX, deltaY);
                if (dist > 0 && (isDraggingSphere || isDraggingMaster)) {
                    const rotateAngle = 0.005;
                    const axis = new THREE.Vector3(deltaY, deltaX, 0).normalize();
                    axis.transformDirection(camera.matrixWorld);
                    const q = new THREE.Quaternion().setFromAxisAngle(axis, dist * rotateAngle);

                    if (!isFlattened && isDraggingSphere && draggedGroup) {
                        draggedGroup.quaternion.premultiply(q);
                        draggedGroup.userData.baseQuaternion.copy(draggedGroup.quaternion);
                    } else if (isDraggingMaster) {
                        masterTargetQuat.premultiply(q);
                    }
                }
                previousMousePosition = { x: event.clientX, y: event.clientY };
            });

            window.addEventListener('pointerup', (event) => {
                // 단순 클릭인지(드래그가 아닌지) 검사
                const distMoved = Math.hypot(event.clientX - pointerDownPos.x, event.clientY - pointerDownPos.y);
                
                if (distMoved < 5) { // 5px 이내의 움직임은 클릭으로 간주
                    raycaster.setFromCamera(mouse, camera);
                    const intersects = raycaster.intersectObjects(hitboxesArray);
                    if (intersects.length > 0) {
                        const topic = intersects[0].object.userData.topicName;
                        if (topic) openEditor(topic);
                    }
                }

                isDraggingSphere = false;
                isDraggingMaster = false;
                draggedGroup = null;
                container.style.cursor = '';
            });

            // 반응형 리사이즈 처리 (에디터 패널 오픈 시 애니메이션과 자연스럽게 동기화)
            const resizeObserver = new ResizeObserver(() => {
                window.requestAnimationFrame(() => {
                    canvasWidth = graphContainer.clientWidth;
                    canvasHeight = graphContainer.clientHeight;
                    
                    if(canvasWidth === 0 || canvasHeight === 0) return;
                    
                    camera.aspect = canvasWidth / canvasHeight;
                    camera.updateProjectionMatrix();
                    renderer.setSize(canvasWidth, canvasHeight);
                });
            });
            resizeObserver.observe(graphContainer);

            function animate() {
                requestAnimationFrame(animate);
                
                camera.position.z += (targetZoomZ - camera.position.z) * 0.05;
                const currentTargetNavZ = isFlattened ? targetNavZ : 0;
                masterTranslationGroup.position.z += (currentTargetNavZ - masterTranslationGroup.position.z) * 0.05;
                masterRotationGroup.quaternion.slerp(masterTargetQuat, 0.08);

                const targetProgress = isFlattened ? 1.0 : 0.0;
                globalFlattenProgress += (targetProgress - globalFlattenProgress) * 0.03;

                const targetFogNear = isFlattened ? targetZoomZ - 10 : 20;
                const targetFogFar = isFlattened ? targetZoomZ + 80 : 120;
                scene.fog.near += (targetFogNear - scene.fog.near) * 0.05;
                scene.fog.far += (targetFogFar - scene.fog.far) * 0.05;

                if (Math.abs(globalFlattenProgress - targetProgress) > 0.001) {
                    groupUpdaters.forEach(updater => updater(globalFlattenProgress));
                }

                particlesMesh.rotation.y += 0.0003; particlesMesh.rotation.x += 0.0001;

                if (globalFlattenProgress < 0.01) {
                    allSphereGroups.forEach(group => {
                        if (group !== draggedGroup) {
                            group.rotateOnWorldAxis(new THREE.Vector3(0,1,0), 0.0008);
                            group.rotateOnWorldAxis(new THREE.Vector3(1,0,0), 0.0005);
                            group.userData.baseQuaternion.copy(group.quaternion); 
                        }
                    });
                }

                const fiberPosAttr = fiberGeometry.attributes.position;
                let fiberIdx = 0;
                interSphereConnections.forEach(conn => {
                    tempPosA.copy(conn.nodeA.position).applyMatrix4(conn.nodeA.parent.matrix);
                    tempPosB.copy(conn.nodeB.position).applyMatrix4(conn.nodeB.parent.matrix);
                    const dist = tempPosA.distanceTo(tempPosB);
                    const midPoint = tempPosA.clone().lerp(tempPosB, 0.5);
                    const outwardForce = midPoint.clone().normalize().multiplyScalar(dist * 0.15);
                    const twistForce = conn.randomDir.clone().multiplyScalar(dist * 0.1);
                    
                    tempFiberCurve.v0.copy(tempPosA);
                    tempFiberCurve.v1.copy(midPoint).add(outwardForce).add(twistForce);
                    tempFiberCurve.v2.copy(tempPosB);

                    const points = tempFiberCurve.getPoints(segmentsPerFiber);
                    for (let i = 0; i < segmentsPerFiber; i++) {
                        fiberPosAttr.setXYZ(fiberIdx++, points[i].x, points[i].y, points[i].z);
                        fiberPosAttr.setXYZ(fiberIdx++, points[i+1].x, points[i+1].y, points[i+1].z);
                    }
                });
                fiberPosAttr.needsUpdate = true;

                // 호버링 로직 분리 (작은 노드들에만 스케일 효과 적용)
                raycaster.setFromCamera(mouse, camera);
                if (!isDraggingSphere && !isDraggingMaster) {
                    const intersects = raycaster.intersectObjects(nodesArray);
                    if (intersects.length > 0) {
                        const object = intersects[0].object;
                        if (hoveredNode !== object) {
                            if (hoveredNode) { hoveredNode.scale.setScalar(hoveredNode.userData.baseScale); hoveredNode.material.color.setHex(hoveredNode.userData.baseColor); }
                            hoveredNode = object;
                            hoveredNode.scale.setScalar(hoveredNode.userData.hoverScale); hoveredNode.material.color.setHex(hoveredNode.userData.hoverColor);
                            container.style.cursor = 'pointer';
                        }
                    } else if (hoveredNode) {
                        hoveredNode.scale.setScalar(hoveredNode.userData.baseScale); hoveredNode.material.color.setHex(hoveredNode.userData.baseColor);
                        hoveredNode = null; container.style.cursor = '';
                    }
                }

                renderer.render(scene, camera);
            }

            animate();
        };
    </script>
